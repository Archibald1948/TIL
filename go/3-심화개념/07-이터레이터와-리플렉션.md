# 이터레이터와 리플렉션 - range over func, iter 패키지, reflect

## 들어가며

이 문서는 성격이 다른 두 주제를 다룹니다. 공통점은 **"타입에 상관없이 범용적인 코드를 쓰는 도구"** 라는 점입니다.

```
이터레이터 (Go 1.23+)
- 직접 만든 자료구조도 for range 로 순회
- 전체를 슬라이스로 만들지 않고 필요한 만큼만 지연(lazy) 처리
- JS 의 function* / yield, Symbol.iterator 와 비슷한 역할

리플렉션 (reflect)
- 런타임에 값의 타입, 필드, 태그, 메서드를 조사하고 조작
- encoding/json, ORM, 검증 라이브러리, DI 컨테이너의 기반
- 강력하지만 느리고 타입 안정성이 없음 → 꼭 필요할 때만
```

# 1. range over func (Go 1.23+)

## 이터레이터 = yield 함수를 받는 함수

```go
package main

import "fmt"

// func(yield func(V) bool) 형태의 함수는 for range 로 순회할 수 있다
func Countdown(n int) func(yield func(int) bool) {
	return func(yield func(int) bool) {
		for i := n; i > 0; i-- {
			if !yield(i) { // 루프 쪽에서 break 하면 yield 가 false 반환
				return // → 즉시 멈춰야 함
			}
		}
	}
}

func main() {
	for v := range Countdown(5) {
		if v == 2 {
			break
		}
		fmt.Print(v, " ")
	}
	fmt.Println()
	// 출력: 5 4 3
}
```

```
동작 원리:
for v := range seq { body }

는 컴파일러가 대략 이렇게 바꿔준다:

seq(func(v int) bool {
	body
	return true   // continue → true, break → false
})
```

```ts
// JS 제너레이터와 비교
function* countdown(n) {
  for (let i = n; i > 0; i--) yield i;
}
for (const v of countdown(5)) { ... }
```

## iter.Seq 와 iter.Seq2

표준 라이브러리는 이터레이터 함수 타입에 이름을 붙여 두었습니다.

```go
package iter

type Seq[V any] func(yield func(V) bool)        // 값 하나
type Seq2[K, V any] func(yield func(K, V) bool) // 키-값 쌍 (인덱스-값, 키-값)
```

```go
package main

import (
	"fmt"
	"iter"
)

// 짝수만 무한히 생성 (지연 평가라 무한도 가능)
func Evens() iter.Seq[int] {
	return func(yield func(int) bool) {
		for i := 0; ; i += 2 {
			if !yield(i) {
				return
			}
		}
	}
}

// 인덱스와 함께 반환하는 Seq2
func Enumerate[T any](items []T) iter.Seq2[int, T] {
	return func(yield func(int, T) bool) {
		for i, v := range items {
			if !yield(i, v) {
				return
			}
		}
	}
}

func main() {
	for n := range Evens() {
		if n > 8 {
			break
		}
		fmt.Print(n, " ")
	}
	fmt.Println()
	// 출력: 0 2 4 6 8

	for i, s := range Enumerate([]string{"go", "rust"}) {
		fmt.Println(i, s)
	}
	// 출력:
	// 0 go
	// 1 rust
}
```

## yield가 false를 반환한 뒤 또 호출하면 panic

```go
package main

import "fmt"

func bad(yield func(int) bool) {
	yield(1)
	yield(2) // ❌ false 확인 없이 계속 호출
}

func main() {
	defer func() { fmt.Println("recover:", recover()) }()
	for v := range bad {
		fmt.Println(v)
		break
	}
	// 출력:
	// 1
	// recover: runtime error: range function continued iteration after function for loop body returned false
}
```

# 2. 직접 만든 자료구조에 이터레이터 달기

```go
package main

import (
	"fmt"
	"iter"
)

type Tree[T any] struct {
	Left, Right *Tree[T]
	Value       T
}

// 중위 순회 이터레이터 (재귀도 자연스럽게 표현됨)
func (t *Tree[T]) All() iter.Seq[T] {
	return func(yield func(T) bool) {
		t.push(yield)
	}
}

func (t *Tree[T]) push(yield func(T) bool) bool {
	if t == nil {
		return true
	}
	return t.Left.push(yield) && yield(t.Value) && t.Right.push(yield)
}

func main() {
	//       4
	//     2   6
	//    1 3 5 7
	tree := &Tree[int]{
		Value: 4,
		Left:  &Tree[int]{Value: 2, Left: &Tree[int]{Value: 1}, Right: &Tree[int]{Value: 3}},
		Right: &Tree[int]{Value: 6, Left: &Tree[int]{Value: 5}, Right: &Tree[int]{Value: 7}},
	}

	for v := range tree.All() {
		fmt.Print(v, " ")
		if v == 5 {
			break // 나머지 노드는 방문하지 않음
		}
	}
	fmt.Println()
	// 출력: 1 2 3 4 5
}
```

```
이터레이터 메서드 이름 관례:
All()     → 전체 원소 (Seq 또는 Seq2)
Keys()    → 키만
Values()  → 값만
Backward() → 역순
```

# 3. 표준 라이브러리의 이터레이터

```go
package main

import (
	"fmt"
	"maps"
	"slices"
	"strings"
)

func main() {
	s := []string{"c", "a", "b"}

	// slices → 이터레이터
	for i, v := range slices.All(s) { // Seq2[int, string]
		fmt.Print(i, v, " ")
	}
	fmt.Println()

	for v := range slices.Backward(s) { // 역순 (인덱스, 값)
		_ = v
	}

	// 이터레이터 → slices
	sorted := slices.Sorted(slices.Values(s))
	fmt.Println(sorted) // [a b c]

	m := map[string]int{"x": 1, "y": 2, "z": 3}
	keys := slices.Sorted(maps.Keys(m)) // 맵 키를 정렬된 슬라이스로
	fmt.Println(keys)                   // [x y z]

	// 이터레이터 → map
	m2 := maps.Collect(maps.All(m))
	fmt.Println(len(m2)) // 3

	// 문자열 이터레이터 (Go 1.24+): 중간 슬라이스를 만들지 않음
	text := "첫 줄\n둘째 줄\n셋째 줄"
	for line := range strings.Lines(text) {
		fmt.Printf("%q\n", line)
	}
	// 출력:
	// "첫 줄\n"
	// "둘째 줄\n"
	// "셋째 줄"

	for field := range strings.FieldsSeq("  go   is  fun ") {
		fmt.Print("[", field, "]")
	}
	fmt.Println()
	// 출력: [go][is][fun]

	// slices.Chunk: n 개씩 묶기
	for chunk := range slices.Chunk([]int{1, 2, 3, 4, 5}, 2) {
		fmt.Print(chunk, " ")
	}
	fmt.Println()
	// 출력: [1 2] [3 4] [5]
}
```

## 이터레이터 조합: Map, Filter, Take 직접 만들기

표준 라이브러리에는 아직 `Map`/`Filter`가 없지만 쉽게 만들 수 있습니다.

```go
package main

import (
	"fmt"
	"iter"
	"slices"
)

func Filter[V any](seq iter.Seq[V], pred func(V) bool) iter.Seq[V] {
	return func(yield func(V) bool) {
		for v := range seq {
			if pred(v) && !yield(v) {
				return
			}
		}
	}
}

func Map[V, R any](seq iter.Seq[V], f func(V) R) iter.Seq[R] {
	return func(yield func(R) bool) {
		for v := range seq {
			if !yield(f(v)) {
				return
			}
		}
	}
}

func Take[V any](seq iter.Seq[V], n int) iter.Seq[V] {
	return func(yield func(V) bool) {
		if n <= 0 {
			return
		}
		count := 0
		for v := range seq {
			if !yield(v) {
				return
			}
			count++
			if count == n {
				return
			}
		}
	}
}

func Naturals() iter.Seq[int] {
	return func(yield func(int) bool) {
		for i := 1; ; i++ {
			if !yield(i) {
				return
			}
		}
	}
}

func main() {
	// 무한 수열 → 3의 배수 → 제곱 → 앞에서 5개
	// 필요한 만큼만 계산됨 (lazy)
	result := slices.Collect(
		Take(Map(Filter(Naturals(), func(n int) bool { return n%3 == 0 }),
			func(n int) int { return n * n }), 5),
	)
	fmt.Println(result)
	// 출력: [9 36 81 144 225]
}
```

# 4. Pull 이터레이터

지금까지의 이터레이터는 **push 방식**(이터레이터가 값을 밀어줌)입니다. 두 시퀀스를 번갈아 읽는 것처럼 **호출하는 쪽이 원할 때 하나씩 꺼내야** 하면 `iter.Pull`을 씁니다.

```go
package main

import (
	"fmt"
	"iter"
	"slices"
)

// 두 정렬된 시퀀스가 같은지 비교
func EqualSeq[V comparable](a, b iter.Seq[V]) bool {
	nextA, stopA := iter.Pull(a)
	defer stopA() // 반드시 호출해서 내부 고루틴(코루틴) 정리
	nextB, stopB := iter.Pull(b)
	defer stopB()

	for {
		va, okA := nextA()
		vb, okB := nextB()
		if okA != okB || va != vb {
			return false
		}
		if !okA {
			return true
		}
	}
}

func main() {
	a := slices.Values([]int{1, 2, 3})
	b := slices.Values([]int{1, 2, 3})
	c := slices.Values([]int{1, 2, 4})
	fmt.Println(EqualSeq(a, b), EqualSeq(a, c))
	// 출력: true false
}
```

# 5. 리플렉션 기초

## TypeOf와 ValueOf

```go
package main

import (
	"fmt"
	"reflect"
)

type User struct {
	Name string
	Age  int
}

func inspect(v any) {
	t := reflect.TypeOf(v)    // 타입 정보
	val := reflect.ValueOf(v) // 값 정보

	fmt.Printf("Type=%v Kind=%v Value=%v\n", t, t.Kind(), val)
}

func main() {
	inspect(42)
	inspect("go")
	inspect([]int{1, 2})
	inspect(User{"alice", 30})
	inspect(&User{})
	// 출력:
	// Type=int Kind=int Value=42
	// Type=string Kind=string Value=go
	// Type=[]int Kind=slice Value=[1 2]
	// Type=main.User Kind=struct Value={alice 30}
	// Type=*main.User Kind=ptr Value=&{ 0}
}
```

```
Type → 구체적인 타입 이름 (main.User, []int, map[string]int)
Kind → 타입의 "종류" (struct, slice, map, ptr, int ...)
       → 리플렉션 코드는 보통 Kind 로 분기함

reflect.TypeFor[T]() (Go 1.22+) → 값 없이 타입 정보 얻기
  reflect.TypeFor[error]()  (예전: reflect.TypeOf((*error)(nil)).Elem())
```

## 구조체 필드와 태그 읽기

```go
package main

import (
	"fmt"
	"reflect"
)

type Product struct {
	ID    int     `json:"id" db:"product_id"`
	Name  string  `json:"name" validate:"required"`
	Price float64 `json:"price" validate:"min=0"`
	stock int     // 비공개 필드
}

func main() {
	p := Product{ID: 1, Name: "키보드", Price: 59000, stock: 3}
	t := reflect.TypeOf(p)
	v := reflect.ValueOf(p)

	for i := range t.NumField() {
		field := t.Field(i)
		fmt.Printf("%-6s %-8v json=%-6q db=%-12q exported=%v value=%v\n",
			field.Name,
			field.Type,
			field.Tag.Get("json"),
			field.Tag.Get("db"),
			field.IsExported(),
			v.Field(i),
		)
	}

	// Go 1.26+: 이터레이터 버전
	for field, value := range reflect.ValueOf(p).Fields() {
		if field.IsExported() {
			_ = value.Interface()
		}
	}
}
```

```
ID     int      json="id"   db="product_id" exported=true value=1
Name   string   json="name" db=""           exported=true value=키보드
Price  float64  json="price" db=""           exported=true value=59000
stock  int      json=""     db=""           exported=false value=3
```

## 값 수정하기: 포인터와 Elem

```go
package main

import (
	"fmt"
	"reflect"
)

type Config struct {
	Host string
	Port int
}

func main() {
	cfg := Config{Host: "localhost", Port: 80}

	// ❌ 값을 넘기면 복사본이라 수정 불가
	v := reflect.ValueOf(cfg)
	fmt.Println(v.Field(0).CanSet())
	// 출력: false

	// ✅ 포인터를 넘기고 Elem() 으로 가리키는 값에 접근
	pv := reflect.ValueOf(&cfg).Elem()
	pv.FieldByName("Host").SetString("example.com")
	pv.FieldByName("Port").SetInt(443)

	fmt.Printf("%+v\n", cfg)
	// 출력: {Host:example.com Port:443}
}
```

import (

	"reflect"

type Calculator struct{}

func (Calculator) Add(a, b int) int { return a + b }

# 6. 실전 예제: 태그 기반 검증기

`go-playground/validator` 같은 라이브러리가 내부에서 하는 일을 간단하게 흉내 내봅니다.

```go
package main

import (
	"errors"
	"fmt"
	"reflect"
	"strconv"
	"strings"
)

func Validate(s any) error {
	v := reflect.ValueOf(s)
	if v.Kind() == reflect.Pointer {
		v = v.Elem()
	}
	if v.Kind() != reflect.Struct {
		return errors.New("구조체만 검증할 수 있음")
	}

	var errs []error
	t := v.Type()
	for i := range t.NumField() {
		field := t.Field(i)
		tag := field.Tag.Get("validate")
		if tag == "" {
			continue
		}
		fv := v.Field(i)

		for rule := range strings.SplitSeq(tag, ",") {
			name, arg, _ := strings.Cut(rule, "=")
			switch name {
			case "required":
				if fv.IsZero() {
					errs = append(errs, fmt.Errorf("%s: 필수 값", field.Name))
				}
			case "min":
				limit, _ := strconv.Atoi(arg)
				switch fv.Kind() {
				case reflect.Int, reflect.Int64:
					if fv.Int() < int64(limit) {
						errs = append(errs, fmt.Errorf("%s: %d 이상이어야 함", field.Name, limit))
					}
				case reflect.String:
					if len([]rune(fv.String())) < limit {
						errs = append(errs, fmt.Errorf("%s: %d글자 이상이어야 함", field.Name, limit))
					}
				}
			}
		}
	}
	return errors.Join(errs...)
}

type SignUp struct {
	Email    string `validate:"required"`
	Password string `validate:"required,min=8"`
	Age      int    `validate:"min=14"`
}

func main() {
	err := Validate(SignUp{Email: "", Password: "1234", Age: 10})
	fmt.Println(err)
	// 출력:
	// Email: 필수 값
	// Password: 8글자 이상이어야 함
	// Age: 14 이상이어야 함

	fmt.Println(Validate(&SignUp{Email: "a@b.c", Password: "12345678", Age: 20}))
	// 출력: <nil>
}
```

# 7. 리플렉션은 언제 쓸까

```
단점:
❌ 느림: 일반 코드보다 수십 배 느릴 수 있음, 할당도 많음
❌ 타입 안정성 없음: 실수하면 컴파일이 아니라 런타임 panic
❌ 읽기 어려움

Rob Pike 의 격언:
"Clear is better than clever. Reflection is never clear."
```

```
대안을 먼저 고려:
1. 인터페이스           → 동작이 필요할 뿐이라면
2. 제네릭               → 타입만 다르고 로직이 같다면
3. 타입 스위치          → 다룰 타입이 몇 개로 정해져 있다면
4. 코드 생성 (go generate) → 반복되는 보일러플레이트라면
   예) stringer, sqlc, mockgen, protoc-gen-go, easyjson

리플렉션이 적합한 경우:
✅ 직렬화/역직렬화 라이브러리 (JSON, YAML, DB 스캔)
✅ 태그 기반 검증, 설정 파싱, CLI 플래그 바인딩
✅ 테스트 유틸리티 (깊은 비교 등)
✅ 프레임워크 수준의 범용 도구
```

## go generate로 코드 생성

```go
//go:generate go tool stringer -type=Status

type Status int

const (
	Pending Status = iota
	Active
	Closed
)
```

```bash
go get -tool golang.org/x/tools/cmd/stringer
go generate ./...   # status_string.go 생성 → Status.String() 메서드
```

런타임 리플렉션 대신 **컴파일 전에 코드를 만들어 두는 방식**이라 빠르고 타입 안전합니다.

## 정리

- Go 1.23+ `func(yield func(V) bool)` 형태의 함수는 `for range`로 순회 가능
- `iter.Seq[V]`, `iter.Seq2[K, V]`로 이터레이터 타입 표현, yield가 false면 즉시 return
- 이터레이터는 지연 평가라 무한 수열, 조기 종료, 중간 슬라이스 없는 처리에 유리
- `slices.All/Values/Collect/Sorted/Chunk`, `maps.Keys/All/Collect`, `strings.Lines/SplitSeq`(1.24+)
- 호출자가 하나씩 꺼내야 하면 `iter.Pull`, 반드시 `stop()` 호출
- `reflect.TypeOf`/`ValueOf`, `Kind`로 분기, 태그는 `Field(i).Tag.Get`
- 값을 수정하려면 포인터를 넘기고 `Elem()`, `CanSet()` 확인
- 리플렉션은 느리고 불명확하므로 인터페이스 → 제네릭 → 코드 생성을 먼저 고려
