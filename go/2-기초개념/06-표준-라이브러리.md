# 표준 라이브러리 - 프레임워크 없이도 서버를 만드는 힘

## 들어가며

Node.js로 API 서버를 만들려면 보통 이런 패키지를 설치합니다.

```
express, axios, dotenv, winston, dayjs, lodash, uuid ...
```

Go는 **표준 라이브러리만으로 프로덕션 수준의 HTTP 서버, JSON 처리, 로깅, 암호화, 테스트**까지 가능합니다. "batteries included"라는 표현이 잘 어울립니다.

이 문서에서는 실무에서 매일 쓰는 패키지를 정리합니다.

```
fmt            → 포맷 출력
strings        → 문자열 처리
strconv        → 문자열 ↔ 숫자 변환
os             → 파일, 환경 변수, 프로세스
io, bufio      → 스트림 입출력
encoding/json  → JSON
net/http       → HTTP 서버/클라이언트
time           → 시간, 타이머
slices, maps   → 컬렉션 유틸 (1.21+)
log/slog       → 구조화 로깅 (1.21+)
```

# 2. strings

```go
package main

import (
	"fmt"
	"strings"
)

func main() {
	s := "  Hello, Go World  "

	fmt.Println(strings.TrimSpace(s))                  // "Hello, Go World"
	fmt.Println(strings.ToUpper("go"))                 // "GO"
	fmt.Println(strings.Contains(s, "Go"))             // true
	fmt.Println(strings.HasPrefix("golang", "go"))     // true
	fmt.Println(strings.HasSuffix("main.go", ".go"))   // true
	fmt.Println(strings.Index("chicken", "ken"))       // 4
	fmt.Println(strings.Replace("aaa", "a", "b", 2))   // "bba"
	fmt.Println(strings.ReplaceAll("aaa", "a", "b"))   // "bbb"
	fmt.Println(strings.Split("a,b,c", ","))           // [a b c]
	fmt.Println(strings.Join([]string{"a", "b"}, "-")) // "a-b"
	fmt.Println(strings.Fields(" a  b   c "))          // [a b c] (공백 기준 분리)
	fmt.Println(strings.Repeat("ab", 3))               // "ababab"
	fmt.Println(strings.EqualFold("Go", "GO"))         // true (대소문자 무시 비교)

	// Cut: 구분자 기준으로 앞/뒤 나누기 (Go 1.18+)
	key, value, found := strings.Cut("host=localhost", "=")
	fmt.Println(key, value, found) // host localhost true

	// CutLast: 마지막 구분자 기준 (Go 1.27+)
	dir, file, _ := strings.CutLast("a/b/c.txt", "/")
	fmt.Println(dir, file) // a/b c.txt

	// 이터레이터 버전 (Go 1.24+): 슬라이스를 만들지 않고 순회
	for part := range strings.SplitSeq("x,y,z", ",") {
		fmt.Print(part, " ")
	}
	fmt.Println()
}
```

# 3. strconv

```go
package main

import (
	"fmt"
	"strconv"
)

func main() {
	// 문자열 → 숫자
	n, err := strconv.Atoi("42")
	fmt.Println(n, err) // 42 <nil>

	_, err = strconv.Atoi("42abc")
	fmt.Println(err) // strconv.Atoi: parsing "42abc": invalid syntax

	i64, _ := strconv.ParseInt("ff", 16, 64) // 16진수, 64비트
	f, _ := strconv.ParseFloat("3.14", 64)
	b, _ := strconv.ParseBool("true")
	fmt.Println(i64, f, b) // 255 3.14 true

	// 숫자 → 문자열
	fmt.Println(strconv.Itoa(42))                         // "42"
	fmt.Println(strconv.FormatInt(255, 2))                // "11111111"
	fmt.Println(strconv.FormatFloat(3.14159, 'f', 2, 64)) // "3.14"

	// 따옴표 처리
	fmt.Println(strconv.Quote("hi\n")) // "hi\n" (따옴표와 이스케이프가 포함된 문자열)
}
```

# 4. os

```go
package main

import (
	"fmt"
	"os"
	"path/filepath"
)

func main() {
	// 환경 변수
	home := os.Getenv("HOME")
	port, ok := os.LookupEnv("PORT") // 설정 여부까지 확인
	fmt.Println(home != "", port, ok)

	// 파일 한 번에 읽기/쓰기
	dir := os.TempDir()
	path := filepath.Join(dir, "demo.txt") // OS에 맞는 경로 구분자 사용
	if err := os.WriteFile(path, []byte("hello\nworld\n"), 0o644); err != nil {
		fmt.Println(err)
		return
	}
	data, err := os.ReadFile(path)
	if err != nil {
		fmt.Println(err)
		return
	}
	fmt.Print(string(data))

	// 파일 존재 확인
	if _, err := os.Stat(path); os.IsNotExist(err) {
		fmt.Println("없음")
	}

	// 디렉터리
	os.MkdirAll(filepath.Join(dir, "a", "b"), 0o755)
	entries, _ := os.ReadDir(dir)
	fmt.Println(len(entries) > 0)

	os.Remove(path)

	// 명령줄 인자
	fmt.Println(os.Args[0] != "") // 실행 파일 경로, 인자는 os.Args[1:]

	// 종료 코드
	// os.Exit(1)
}
```

```
os.Open(path)                        → 읽기 전용
os.Create(path)                      → 쓰기 (없으면 생성, 있으면 비움)
os.OpenFile(path, flag, perm)        → 세밀한 제어
  예) os.OpenFile("app.log", os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0o644)
```

> 사용자 입력 경로로 파일에 접근한다면 Go 1.24+ `os.OpenRoot`로 특정 디렉터리 밖(`../`)으로 벗어나지 못하게 막을 수 있습니다.

명령줄 플래그는 `flag` 패키지를 씁니다.

```go
port := flag.Int("port", 8080, "서버 포트")
debug := flag.Bool("debug", false, "디버그 모드")
flag.Parse()
// ./app -port 3000 -debug
```

# 5. io 와 bufio

## io.Reader / io.Writer 조합

```go
package main

import (
	"compress/gzip"
	"crypto/sha256"
	"fmt"
	"io"
	"os"
	"strings"
)

func main() {
	src := strings.NewReader("hello, io")

	// 1) 복사: Reader → Writer
	io.Copy(os.Stdout, src)
	fmt.Println()

	// 2) 여러 Writer에 동시에 쓰기 (tee)
	hasher := sha256.New()
	var sb strings.Builder
	w := io.MultiWriter(hasher, &sb)
	io.WriteString(w, "data")
	fmt.Printf("%s %x\n", sb.String(), hasher.Sum(nil)[:4])

	// 3) 압축 스트림도 그냥 Writer
	gz := gzip.NewWriter(io.Discard)
	io.Copy(gz, strings.NewReader(strings.Repeat("a", 1000)))
	gz.Close()

	// 4) 크기 제한 (요청 바디 제한 등에 유용)
	limited := io.LimitReader(strings.NewReader("1234567890"), 3)
	b, _ := io.ReadAll(limited)
	fmt.Println(string(b)) // 123
}
```

## bufio: 줄 단위 읽기

```go
package main

import (
	"bufio"
	"fmt"
	"os"
	"strings"
)

func main() {
	input := "첫 줄\n둘째 줄\n셋째 줄"
	scanner := bufio.NewScanner(strings.NewReader(input)) // 파일이면 os.Open 결과를 넣으면 됨

	lineNo := 0
	for scanner.Scan() {
		lineNo++
		fmt.Printf("%d: %s\n", lineNo, scanner.Text())
	}
	if err := scanner.Err(); err != nil { // 루프 끝나고 에러 확인
		fmt.Println(err)
	}

	// 버퍼링된 쓰기: 작은 쓰기가 많을 때 시스템 콜을 줄여줌
	w := bufio.NewWriter(os.Stdout)
	defer w.Flush() // Flush 안 하면 출력이 안 될 수 있음!
	fmt.Fprintln(w, "버퍼링된 출력")
}
```

> `bufio.Scanner`는 기본적으로 한 줄 최대 64KB입니다. 더 긴 줄이 있으면 `scanner.Buffer(buf, maxSize)`로 늘려야 합니다.

# 6. encoding/json

## 구조체 ↔ JSON

```go
package main

import (
	"encoding/json"
	"fmt"
	"time"
)

type User struct {
	ID        int       `json:"id"`
	Name      string    `json:"name"`
	Email     string    `json:"email,omitempty"`
	Tags      []string  `json:"tags"`
	CreatedAt time.Time `json:"created_at"`
	password  string    // 소문자 → JSON 에서 무시
}

func main() {
	u := User{ID: 1, Name: "gopher", CreatedAt: time.Date(2026, 9, 14, 0, 0, 0, 0, time.UTC)}

	// 인코딩
	b, err := json.Marshal(u)
	if err != nil {
		panic(err)
	}
	fmt.Println(string(b))
	// 출력: {"id":1,"name":"gopher","tags":null,"created_at":"2026-09-14T00:00:00Z"}

	pretty, _ := json.MarshalIndent(u, "", "  ")
	fmt.Println(string(pretty))

	// 디코딩
	var decoded User
	input := `{"id": 2, "name": "rustacean", "tags": ["a", "b"], "unknown": 1}`
	if err := json.Unmarshal([]byte(input), &decoded); err != nil {
		panic(err)
	}
	fmt.Printf("%+v\n", decoded.Tags)
	// 출력: [a b]    (모르는 필드 "unknown" 은 조용히 무시됨)
}
```

## 구조를 모를 때

```go
package main

import (
	"encoding/json"
	"fmt"
)

func main() {
	var data map[string]any
	json.Unmarshal([]byte(`{"a": 1, "b": [true, "x"]}`), &data)

	// JSON 숫자는 float64 로 들어옴!
	fmt.Printf("%T %T\n", data["a"], data["b"])
	// 출력: float64 []interface {}

	// 일부만 나중에 파싱: json.RawMessage
	var envelope struct {
		Type    string          `json:"type"`
		Payload json.RawMessage `json:"payload"`
	}
	json.Unmarshal([]byte(`{"type":"user","payload":{"id":1}}`), &envelope)
	fmt.Println(envelope.Type, string(envelope.Payload))
	// 출력: user {"id":1}
}
```

## encoding/json/v2 (Go 1.27+)

Go 1.27에서 기존 json의 여러 문제를 고친 **v2가 정식으로 추가**되었습니다.

```go
package main

import (
	"encoding/json/v2"
	"fmt"
)

type Item struct {
	Name string   `json:"name"`
	Tags []string `json:"tags"`
}

func main() {
	b, _ := json.Marshal(Item{Name: "go"})
	fmt.Println(string(b))
	// 출력: {"name":"go","tags":[]}    ← v1은 "tags":null

	var it Item
	err := json.Unmarshal([]byte(`{"NAME":"x"}`), &it)
	fmt.Printf("%q %v\n", it.Name, err)
	// 출력: "" <nil>    ← v2는 필드 이름을 대소문자 구분해서 매칭 (v1은 무시하고 매칭)
}
```

```
v1 → v2 주요 변경:
- nil 슬라이스/맵을 [] / {} 로 인코딩 (v1: null)
- 필드 이름 대소문자 구분 매칭
- 중복 키, 잘못된 UTF-8 거부
- Options 로 동작 제어 (json.RejectUnknownMembers(true) 등)
- 저수준 토큰 처리는 encoding/json/jsontext 패키지
```

기존 `encoding/json`(v1)도 계속 지원되므로 새 코드부터 천천히 적용하면 됩니다.

# 7. net/http

## 서버 (Go 1.22+ 라우팅)

Go 1.22부터 `ServeMux`가 **HTTP 메서드와 경로 파라미터**를 지원해서, 간단한 API는 프레임워크 없이도 충분합니다.

```go
package main

import (
	"encoding/json"
	"log"
	"net/http"
	"strconv"
	"sync"
)

type Todo struct {
	ID    int    `json:"id"`
	Title string `json:"title"`
	Done  bool   `json:"done"`
}

type store struct {
	mu     sync.Mutex
	todos  map[int]Todo
	nextID int
}

func writeJSON(w http.ResponseWriter, status int, v any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	json.NewEncoder(w).Encode(v)
}

func main() {
	s := &store{todos: map[int]Todo{}, nextID: 1}
	mux := http.NewServeMux()

	// "메서드 경로" 패턴
	mux.HandleFunc("GET /todos", func(w http.ResponseWriter, r *http.Request) {
		s.mu.Lock()
		defer s.mu.Unlock()
		list := make([]Todo, 0, len(s.todos))
		for _, t := range s.todos {
			list = append(list, t)
		}
		writeJSON(w, http.StatusOK, list)
	})

	mux.HandleFunc("POST /todos", func(w http.ResponseWriter, r *http.Request) {
		var t Todo
		if err := json.NewDecoder(r.Body).Decode(&t); err != nil {
			http.Error(w, "잘못된 JSON", http.StatusBadRequest)
			return
		}
		s.mu.Lock()
		t.ID = s.nextID
		s.nextID++
		s.todos[t.ID] = t
		s.mu.Unlock()
		writeJSON(w, http.StatusCreated, t)
	})

	// {id} 경로 파라미터
	mux.HandleFunc("GET /todos/{id}", func(w http.ResponseWriter, r *http.Request) {
		id, err := strconv.Atoi(r.PathValue("id"))
		if err != nil {
			http.Error(w, "잘못된 id", http.StatusBadRequest)
			return
		}
		s.mu.Lock()
		t, ok := s.todos[id]
		s.mu.Unlock()
		if !ok {
			http.NotFound(w, r)
			return
		}
		writeJSON(w, http.StatusOK, t)
	})

	log.Println("http://localhost:8080 에서 실행 중")
	log.Fatal(http.ListenAndServe(":8080", mux))
}
```

```bash
curl -X POST localhost:8080/todos -d '{"title":"Go 공부"}'
# {"id":1,"title":"Go 공부","done":false}

curl localhost:8080/todos/1
curl -X DELETE localhost:8080/todos/1
# Method Not Allowed (405) ← 메서드가 등록 안 된 경우 자동 처리
```

```
패턴 문법 (Go 1.22+):
"GET /posts/{id}"        → r.PathValue("id")
"/files/{path...}"       → 나머지 경로 전체
"/posts/{$}"             → 정확히 /posts/ 만 (하위 경로 매칭 X)
"GET api.example.com/"   → 호스트 지정
```

> 각 요청은 **별도의 고루틴**에서 처리됩니다. 그래서 위 예제에서 공유 맵을 `sync.Mutex`로 보호했습니다.

## 클라이언트

```go
package main

import (
	"context"
	"encoding/json"
	"fmt"
	"net/http"
	"net/http/httptest"
	"time"
)

func main() {
	// 테스트용 가짜 API 서버
	api := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		fmt.Fprint(w, `{"login":"gopher","id":1}`)
	}))
	defer api.Close()

	// ❌ http.Get 은 기본 클라이언트(타임아웃 없음) 사용
	// ✅ 타임아웃이 있는 클라이언트를 만들어서 재사용
	client := &http.Client{Timeout: 10 * time.Second}

	ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
	defer cancel()

	req, _ := http.NewRequestWithContext(ctx, http.MethodGet, api.URL+"/users/gopher", nil)
	req.Header.Set("Accept", "application/json")

	resp, err := client.Do(req)
	if err != nil {
		fmt.Println(err)
		return
	}
	defer resp.Body.Close() // 반드시 닫기 (커넥션 재사용을 위해)

	if resp.StatusCode != http.StatusOK {
		fmt.Println("status:", resp.Status)
		return
	}

	var user struct {
		Login string `json:"login"`
		ID    int    `json:"id"`
	}
	json.NewDecoder(resp.Body).Decode(&user)
	fmt.Printf("%+v\n", user)
	// 출력: {Login:gopher ID:1}
}
```

# 8. time

```go
package main

import (
	"fmt"
	"time"
)

func main() {
	now := time.Now()

	// Duration: 나노초 단위 int64
	d := 90 * time.Minute
	fmt.Println(d, d.Hours()) // 1h30m0s 1.5

	later := now.Add(2 * time.Hour)
	fmt.Println(later.Sub(now))       // 2h0m0s
	fmt.Println(later.After(now))     // true
	fmt.Println(time.Since(now) >= 0) // true

	// 포맷: "2006-01-02 15:04:05" 라는 기준 시각으로 레이아웃을 표현
	t := time.Date(2026, 9, 14, 21, 5, 0, 0, time.UTC)
	fmt.Println(t.Format("2006-01-02 15:04:05")) // 2026-09-14 21:05:00
	fmt.Println(t.Format(time.RFC3339))          // 2026-09-14T21:05:00Z
	fmt.Println(t.Format("2006년 1월 2일 (Mon)"))   // 2026년 9월 14일 (Mon)

	parsed, err := time.Parse("2006-01-02", "2026-12-25")
	fmt.Println(parsed.Weekday(), err) // Friday <nil>

	// 타임존
	seoul, _ := time.LoadLocation("Asia/Seoul")
	fmt.Println(t.In(seoul).Format("15:04 MST")) // 06:05 KST

	// 경과 시간 측정
	start := time.Now()
	time.Sleep(10 * time.Millisecond)
	fmt.Println(time.Since(start) >= 10*time.Millisecond) // true
}
```

```
레이아웃 기준 시각: Mon Jan 2 15:04:05 MST 2006
                    1    2   3  4  5       6   (월 일 시 분 초 년 → 1 2 3 4 5 6)

YYYY-MM-DD  →  2006-01-02
HH:mm:ss    →  15:04:05
```

## 타이머와 티커

```go
timer := time.NewTimer(2 * time.Second)  // 한 번
ticker := time.NewTicker(1 * time.Second) // 주기적
defer ticker.Stop()

select {
case <-timer.C:
case <-ticker.C:
case <-time.After(5 * time.Second): // 간단한 타임아웃
}
```

# 10. log/slog (구조화 로깅, Go 1.21+)

```go
package main

import (
	"log/slog"
	"os"
)

func main() {
	// 기본: 텍스트 형식
	slog.Info("서버 시작", "port", 8080, "env", "dev")
	// 2026/09/14 21:05:00 INFO 서버 시작 port=8080 env=dev

	// JSON 핸들러 (로그 수집 시스템에 적합)
	logger := slog.New(slog.NewJSONHandler(os.Stdout, &slog.HandlerOptions{
		Level: slog.LevelDebug,
	}))
	slog.SetDefault(logger)

	slog.Debug("디버그", "user_id", 42)
	// {"time":"...","level":"DEBUG","msg":"디버그","user_id":42}

	// 공통 필드를 붙인 로거
	reqLogger := logger.With("request_id", "abc-123")
	reqLogger.Warn("느린 쿼리", "ms", 1200)

	// 그룹
	logger.Error("결제 실패",
		slog.Group("order", slog.Int("id", 7), slog.String("status", "failed")),
	)
	// {"time":"...","level":"ERROR","msg":"결제 실패","order":{"id":7,"status":"failed"}}
}
```

```
log (기존)       → log.Printf("user %d login", id)   단순 문자열
log/slog (1.21+) → slog.Info("login", "user", id)    키-값 구조화, 레벨, JSON 출력
```

> Go 1.26+ `slog.NewMultiHandler`로 여러 핸들러(예: 콘솔 + 파일)에 동시에 기록할 수 있습니다.

## 정리

- 문자열은 `strings`(+ `strings.Builder`), 숫자 변환은 `strconv`
- 파일/환경 변수는 `os`, 경로는 `path/filepath`, 스트림은 `io`/`bufio`
- `io.Reader`/`io.Writer` 인터페이스로 파일·네트워크·압축·해시를 자유롭게 조합
- JSON은 `encoding/json`, Go 1.27+ `encoding/json/v2`에서 기본 동작 개선
- Go 1.22+ `http.ServeMux`는 `"GET /users/{id}"` 패턴과 `r.PathValue` 지원
- HTTP 클라이언트/서버는 타임아웃을 반드시 설정, 응답 바디는 반드시 Close
- 시간 포맷은 `2006-01-02 15:04:05` 기준 시각으로 표현
- 로깅은 `log/slog`로 구조화
