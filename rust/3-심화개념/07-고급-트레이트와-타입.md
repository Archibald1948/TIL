# 1. 기본 제네릭 타입 매개변수와 연산자 오버로딩

`Add` 트레이트의 실제 정의를 보면 `Rhs`에 기본값이 있습니다.

```rust
// (컴파일 검증 제외) - 표준 라이브러리 정의
trait Add<Rhs = Self> {   // Rhs 를 안 적으면 Self
    type Output;
    fn add(self, rhs: Rhs) -> Self::Output;
}
```

그래서 같은 타입끼리 더할 땐 `impl Add for Point`, 다른 타입과 더할 땐 `impl Add<Other> for Point`로 씁니다.

```rust
use std::ops::{Add, AddAssign, Index, Mul, Neg};

#[derive(Debug, Clone, Copy, PartialEq)]
struct Millimeters(u32);
#[derive(Debug, Clone, Copy, PartialEq)]
struct Meters(u32);

// mm + mm
impl Add for Millimeters {
    type Output = Millimeters;
    fn add(self, other: Millimeters) -> Millimeters {
        Millimeters(self.0 + other.0)
    }
}

// mm + m (Rhs 를 Meters 로 지정)
impl Add<Meters> for Millimeters {
    type Output = Millimeters;
    fn add(self, other: Meters) -> Millimeters {
        Millimeters(self.0 + other.0 * 1000)
    }
}

#[derive(Debug, Clone, Copy)]
struct Vec2 {
    x: f64,
    y: f64,
}

impl Mul<f64> for Vec2 {          // Vec2 * f64
    type Output = Vec2;
    fn mul(self, k: f64) -> Vec2 {
        Vec2 { x: self.x * k, y: self.y * k }
    }
}

impl Neg for Vec2 {               // -Vec2
    type Output = Vec2;
    fn neg(self) -> Vec2 {
        Vec2 { x: -self.x, y: -self.y }
    }
}

impl AddAssign for Vec2 {         // +=
    fn add_assign(&mut self, o: Vec2) {
        self.x += o.x;
        self.y += o.y;
    }
}

struct Polynomial(Vec<i32>);

impl Index<usize> for Polynomial { // poly[i]
    type Output = i32;
    fn index(&self, i: usize) -> &i32 {
        &self.0[i]
    }
}

fn main() {
    println!("{:?}", Millimeters(500) + Millimeters(250));
    println!("{:?}", Millimeters(500) + Meters(2));

    let mut v = Vec2 { x: 1.0, y: 2.0 } * 3.0;
    v += -Vec2 { x: 1.0, y: 1.0 };
    println!("{v:?}");

    let p = Polynomial(vec![3, 0, 2]);
    println!("x^2 계수 = {}", p[2]);
}
```

# 2. 완전 정규화 문법 (Fully Qualified Syntax)

여러 트레이트에 같은 이름의 메서드가 있을 때 어느 것을 호출할지 명시합니다.

```rust
trait Pilot {
    fn fly(&self);
    fn name() -> String;
}

trait Wizard {
    fn fly(&self);
    fn name() -> String;
}

struct Human;

impl Pilot for Human {
    fn fly(&self) { println!("기장입니다. 이륙합니다."); }
    fn name() -> String { "조종사".into() }
}

impl Wizard for Human {
    fn fly(&self) { println!("날아올라!"); }
    fn name() -> String { "마법사".into() }
}

impl Human {
    fn fly(&self) { println!("*팔을 격렬하게 흔든다*"); }
    fn name() -> String { "인간".into() }
}

fn main() {
    let person = Human;
    person.fly();          // 타입 자체의 메서드가 우선
    Pilot::fly(&person);   // self 가 있으면 트레이트 이름으로 구분 가능
    Wizard::fly(&person);

    println!("{}", Human::name());
    // println!("{}", Pilot::name()); // ❌ self 가 없어서 어떤 타입의 구현인지 알 수 없음
    println!("{}", <Human as Pilot>::name());  // 완전 정규화 문법
    println!("{}", <Human as Wizard>::name());
}
```

```
<Type as Trait>::function(receiver_if_method, args...)
```

제네릭 코드에서 `<T as Iterator>::Item`처럼 **연관 타입을 명확히 가리킬 때**도 씁니다.

# 3. 확장 트레이트와 봉인 트레이트

## 확장 트레이트 (Extension Trait)

외부 타입에 메서드를 "추가"하는 패턴입니다. 고아 규칙상 트레이트가 내 것이므로 가능합니다. `itertools`의 `Itertools`, `futures`의 `StreamExt`가 이 방식입니다.

```rust
pub trait StrExt {
    fn is_palindrome(&self) -> bool;
    fn word_count(&self) -> usize;
}

impl StrExt for str {
    fn is_palindrome(&self) -> bool {
        let chars: Vec<char> = self.chars().filter(|c| c.is_alphanumeric()).collect();
        chars.iter().eq(chars.iter().rev())
    }
    fn word_count(&self) -> usize {
        self.split_whitespace().count()
    }
}

pub trait IteratorExt: Iterator {
    // 모든 Iterator 에 기본 구현으로 메서드 추가
    fn sum_squares(self) -> i64
    where
        Self: Sized,
        Self::Item: Into<i64>,
    {
        self.map(|x| {
            let v: i64 = x.into();
            v * v
        })
        .sum()
    }
}

impl<I: Iterator> IteratorExt for I {} // 블랭킷 구현

fn main() {
    println!("{}", "never odd or even".is_palindrome());
    println!("{}", String::from("hello rust world").word_count()); // String 은 Deref 로 str 메서드 사용
    println!("{}", vec![1i32, 2, 3].into_iter().sum_squares());
}
```

## 봉인 트레이트 (Sealed Trait)

라이브러리가 공개한 트레이트를 **외부에서는 구현하지 못하게** 막는 패턴입니다. 나중에 트레이트에 메서드를 추가해도 사용자 코드가 깨지지 않습니다.

```rust
mod shapes {
    mod private {
        pub trait Sealed {} // 비공개 모듈 안의 공개 트레이트 → 외부에서 이름을 쓸 수 없음
    }

    pub trait Shape: private::Sealed {
        fn area(&self) -> f64;
    }

    pub struct Circle(pub f64);
    impl private::Sealed for Circle {}
    impl Shape for Circle {
        fn area(&self) -> f64 {
            3.14159 * self.0 * self.0
        }
    }
}

use shapes::Shape;

// 외부 크레이트에서는 private::Sealed 를 구현할 수 없으므로 Shape 도 구현 불가
// struct Square; impl Shape for Square { ... } // ❌

fn main() {
    println!("{:.2}", shapes::Circle(1.0).area());
}
```

# 4. 타입 별칭과 never 타입

## 타입 별칭

```rust
use std::collections::HashMap;
use std::fmt;

// 긴 타입에 이름 붙이기 (새 타입이 아니라 "동의어")
type Thunk = Box<dyn Fn() -> String + Send + 'static>;
type UserMap = HashMap<u64, (String, Vec<String>)>;

// std::io::Result<T> 도 이런 별칭: type Result<T> = std::result::Result<T, io::Error>;
type Result<T> = std::result::Result<T, fmt::Error>;

fn make_thunk(name: &str) -> Thunk {
    let name = name.to_string();
    Box::new(move || format!("hello {name}"))
}

fn write_something(out: &mut String) -> Result<()> {
    use std::fmt::Write;
    write!(out, "written")?;
    Ok(())
}

fn main() {
    let t = make_thunk("alias");
    println!("{}", t());

    let mut users: UserMap = HashMap::new();
    users.insert(1, ("kim".into(), vec!["admin".into()]));
    println!("{users:?}");

    let mut s = String::new();
    write_something(&mut s).unwrap();
    println!("{s}");
}
```

```
type Meters = u32;    → 별칭. u32 와 완전히 같은 타입 (섞어 써도 컴파일됨)
struct Meters(u32);   → 뉴타입. 다른 타입 (실수로 섞으면 컴파일 에러)
```

## never 타입 !

```rust
fn parse_or_exit(s: &str) -> u32 {
    match s.parse::<u32>() {
        Ok(n) => n,
        // continue, break, return, panic!, loop {}, process::exit 는 모두 ! 타입
        // ! 는 어떤 타입으로든 강제 변환되므로 match 갈래 타입이 u32 로 맞춰짐
        Err(_) => {
            eprintln!("숫자가 아님: {s}");
            std::process::exit(1)
        }
    }
}

fn main() {
    let inputs = ["1", "2", "x"];
    let mut total = 0;
    for s in inputs {
        let n: u32 = match s.parse() {
            Ok(n) => n,
            Err(_) => continue, // continue 의 타입은 !
        };
        total += n;
    }
    println!("{total} {}", parse_or_exit("42"));
}
```

# 5. 동적 크기 타입 (DST)과 Sized

컴파일 타임에 크기를 알 수 없는 타입이 있습니다.

```
str       → 문자열 길이는 제각각
[T]       → 슬라이스 길이는 제각각
dyn Trait → 구현 타입마다 크기가 다름
```

이런 타입은 값으로 직접 다룰 수 없고 **항상 포인터 뒤에** 둡니다. 이때 포인터는 추가 정보를 담은 **fat pointer**가 됩니다.

```rust
use std::mem::size_of;

trait Speak {
    fn speak(&self);
}

fn main() {
    let ptr = size_of::<usize>();
    println!("&u8        = {} (포인터)", size_of::<&u8>() / ptr);
    println!("&str       = {} (포인터 + 길이)", size_of::<&str>() / ptr);
    println!("&[i32]     = {} (포인터 + 길이)", size_of::<&[i32]>() / ptr);
    println!("&dyn Speak = {} (포인터 + vtable)", size_of::<&dyn Speak>() / ptr);
    println!("Box<str>   = {}", size_of::<Box<str>>() / ptr);

    // let s: str = *"hello"; // ❌ str 은 크기를 몰라서 지역 변수로 둘 수 없음
}
```

## ?Sized

제네릭 매개변수 `T`에는 암묵적으로 `T: Sized`가 붙어 있습니다. DST도 받으려면 `?Sized`("Sized일 수도 아닐 수도")로 완화합니다.

```rust
use std::fmt::Debug;

// T: Sized (기본) → &str 의 str, &[i32] 의 [i32] 는 T 가 될 수 없음
fn print_sized<T: Debug>(t: &T) {
    println!("{t:?}");
}

// T: ?Sized → str, [T], dyn Trait 도 허용 (반드시 참조나 Box 뒤에서 사용)
fn print_any<T: Debug + ?Sized>(t: &T) {
    println!("{t:?}");
}

fn main() {
    print_sized(&5);
    print_sized(&"literal"); // T = &str (Sized) 라서 OK

    print_any("str 자체"); // T = str
    print_any(&[1, 2, 3][..]); // T = [i32]
    let d: &dyn Debug = &3.14;
    print_any(d); // T = dyn Debug
}
```

`Box<T>`, `Rc<T>`, `Arc<T>` 등이 `T: ?Sized`로 정의되어 있어서 `Box<dyn Trait>`, `Rc<str>`, `Arc<[u8]>`가 가능합니다.

# 6. 함수 포인터와 클로저 반환

```rust
fn add_one(x: i32) -> i32 {
    x + 1
}

// fn 포인터: 크기가 고정된 타입. 캡처 없는 클로저도 변환 가능
fn do_twice(f: fn(i32) -> i32, arg: i32) -> i32 {
    f(f(arg))
}

// 클로저 반환 1: impl Trait (구체 타입 하나)
fn make_multiplier(k: i32) -> impl Fn(i32) -> i32 {
    move |x| x * k
}

// 클로저 반환 2: 분기마다 다른 클로저 → 트레이트 객체
fn make_op(op: &str) -> Box<dyn Fn(i32, i32) -> i32> {
    match op {
        "+" => Box::new(|a, b| a + b),
        "*" => Box::new(|a, b| a * b),
        _ => {
            let fallback = 0;
            Box::new(move |_, _| fallback)
        }
    }
}

#[derive(Debug)]
enum Status {
    Value(u32),
}

fn main() {
    println!("{}", do_twice(add_one, 5));

    // 튜플 구조체/enum variant 의 생성자도 함수처럼 쓸 수 있음
    let statuses: Vec<Status> = (0u32..3).map(Status::Value).collect();
    println!("{statuses:?}");

    let strs: Vec<String> = [1, 2, 3].iter().map(ToString::to_string).collect(); // 메서드 경로
    println!("{strs:?}");

    let triple = make_multiplier(3);
    println!("{}", triple(7));
    println!("{} {}", make_op("+")(2, 3), make_op("*")(2, 3));
}
```

# 7. PhantomData와 타입 상태 패턴

## PhantomData

실제로 값을 저장하지 않지만 **타입 매개변수를 "사용"하는 것처럼** 컴파일러에게 알립니다. 크기는 0입니다.

```rust
use std::marker::PhantomData;

// 단위를 타입으로 구분하는 ID
#[derive(Debug, Clone, Copy, PartialEq)]
struct Id<T> {
    value: u64,
    _marker: PhantomData<T>,
}

impl<T> Id<T> {
    fn new(value: u64) -> Self {
        Id { value, _marker: PhantomData }
    }
}

struct User;
struct Order;

fn find_user(id: Id<User>) -> String {
    format!("user#{}", id.value)
}

fn main() {
    let user_id: Id<User> = Id::new(1);
    let order_id: Id<Order> = Id::new(1);

    println!("{}", find_user(user_id));
    // find_user(order_id); // ❌ mismatched types: Id<Order> 는 Id<User> 가 아님
    println!("크기: {}바이트", std::mem::size_of::<Id<User>>()); // 8 (PhantomData 는 0)
    let _ = order_id;
}
```

## 타입 상태 (Typestate) 패턴

객체의 **상태를 타입으로 표현**해서, 잘못된 순서의 호출을 컴파일 에러로 만듭니다.

```rust
use std::marker::PhantomData;

// 상태 마커 (크기 0)
struct Draft;
struct PendingReview;
struct Published;

struct Post<State> {
    content: String,
    _state: PhantomData<State>,
}

impl Post<Draft> {
    fn new() -> Self {
        Post { content: String::new(), _state: PhantomData }
    }
    fn add_text(&mut self, text: &str) {
        self.content.push_str(text);
    }
    // self 를 소비하고 다른 상태의 타입을 반환 → 이전 상태 객체는 더 이상 사용 불가
    fn request_review(self) -> Post<PendingReview> {
        Post { content: self.content, _state: PhantomData }
    }
}

impl Post<PendingReview> {
    fn approve(self) -> Post<Published> {
        Post { content: self.content, _state: PhantomData }
    }
    fn reject(self) -> Post<Draft> {
        Post { content: self.content, _state: PhantomData }
    }
}

impl Post<Published> {
    // content() 는 Published 상태에만 존재
    fn content(&self) -> &str {
        &self.content
    }
}

fn main() {
    let mut post = Post::new();
    post.add_text("오늘은 러스트를 배웠다");
    // post.content(); // ❌ Post<Draft> 에는 content 메서드가 없음

    let pending = post.request_review();
    // pending.add_text("수정"); // ❌ 리뷰 중에는 수정 불가

    let draft_again = pending.reject();
    let published = draft_again.request_review().approve();
    println!("{}", published.content());
}
```

런타임에 `if state == Published`를 검사하는 대신, **잘못된 호출 자체가 컴파일되지 않습니다.** HTTP 요청 빌더, 파일 핸들(열림/닫힘), 프로토콜 핸드셰이크 같은 곳에 유용합니다.

## 빌더 패턴 + 타입 상태: 필수 필드 강제

```rust
use std::marker::PhantomData;

struct NoUrl;
struct HasUrl;

struct RequestBuilder<U> {
    url: Option<String>,
    headers: Vec<(String, String)>,
    _u: PhantomData<U>,
}

#[derive(Debug)]
struct Request {
    url: String,
    headers: Vec<(String, String)>,
}

impl RequestBuilder<NoUrl> {
    fn new() -> Self {
        RequestBuilder { url: None, headers: vec![], _u: PhantomData }
    }
    fn url(self, url: &str) -> RequestBuilder<HasUrl> {
        RequestBuilder { url: Some(url.into()), headers: self.headers, _u: PhantomData }
    }
}

impl<U> RequestBuilder<U> {
    // 헤더는 어느 상태에서나 추가 가능
    fn header(mut self, k: &str, v: &str) -> Self {
        self.headers.push((k.into(), v.into()));
        self
    }
}

impl RequestBuilder<HasUrl> {
    // build 는 url 을 설정한 뒤에만 호출 가능
    fn build(self) -> Request {
        Request { url: self.url.unwrap(), headers: self.headers }
    }
}

fn main() {
    let req = RequestBuilder::new()
        .header("Accept", "application/json")
        .url("https://example.com")
        .build();
    println!("{req:?}");

    // RequestBuilder::new().header("a", "b").build(); // ❌ url 없이 build 불가
}
```

# 8. 제네릭 연관 타입 (GAT, Rust 1.65+)

연관 타입 자체가 **라이프타임이나 타입 매개변수를 가질** 수 있습니다. 대표 예시는 "빌려주는 이터레이터(lending iterator)"입니다.

```rust
// 표준 Iterator 는 next() 가 반환한 항목이 이터레이터 자신을 빌릴 수 없음
// GAT 를 쓰면 "self 를 빌린 동안만 유효한 항목"을 표현 가능
trait LendingIterator {
    type Item<'a>
    where
        Self: 'a;

    fn next<'a>(&'a mut self) -> Option<Self::Item<'a>>;
}

// 버퍼를 재사용하며 겹치는 윈도우를 가변 슬라이스로 빌려주기
struct WindowsMut<'t, T> {
    slice: &'t mut [T],
    start: usize,
    size: usize,
}

impl<'t, T> LendingIterator for WindowsMut<'t, T> {
    type Item<'a>
        = &'a mut [T]
    where
        Self: 'a;

    fn next<'a>(&'a mut self) -> Option<&'a mut [T]> {
        let end = self.start + self.size;
        if end > self.slice.len() {
            return None;
        }
        let window = &mut self.slice[self.start..end];
        self.start += 1;
        Some(window)
    }
}

fn main() {
    let mut data = [1, 2, 3, 4];
    let mut it = WindowsMut { slice: &mut data, start: 0, size: 2 };
    while let Some(w) = it.next() {
        w[1] += w[0]; // 겹치는 가변 윈도우 (일반 Iterator 로는 불가능)
    }
    println!("{data:?}"); // [1, 3, 6, 10] 누적합
}
```

## 정리

- 연산자 오버로딩은 `std::ops` 트레이트 구현, `Add<Rhs = Self>`처럼 기본 타입 매개변수 활용
- 같은 이름의 메서드는 `Trait::method(&x)` 또는 `<Type as Trait>::func()`로 구분
- 확장 트레이트로 외부 타입에 메서드 추가, 봉인 트레이트로 외부 구현 차단
- `type`은 별칭(같은 타입), 뉴타입은 새 타입
- `str`, `[T]`, `dyn Trait`는 DST라서 포인터 뒤에 두고, 제네릭에서 받으려면 `?Sized`
- `PhantomData`와 타입 상태 패턴으로 잘못된 상태 전이를 컴파일 에러로 만든다
- GAT로 "빌려주는" 연관 타입을 표현할 수 있다
